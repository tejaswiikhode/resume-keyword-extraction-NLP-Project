Theory
An N-Gram Model is a statistical language model used in Natural Language Processing (NLP) to predict the next word in a sequence based on the previous word(s). An N-
gram is a sequence of N consecutive words extracted from a text.
Unigram (N = 1): Single words (e.g., machine, learning)
Bigram (N = 2): Two consecutive words (e.g., machine learning)
Trigram (N = 3): Three consecutive words (e.g., machine learning model)
The N-Gram model works by counting the frequency of word sequences in a dataset and calculating their probabilities. In this experiment, the Resume Dataset is
preprocessed by converting text to lowercase, removing special characters, and tokenizing the text. Unigrams, bigrams, and trigrams are then generated to identify common
word patterns, predict the next word, and analyze language structure.
N-Gram models are widely used in text prediction, autocomplete, spell checking, machine translation, speech recognition, and language modeling. They are simple,
efficient, and serve as the foundation for many NLP applications.

def generate_ngrams(text, n):
    if not isinstance(text, str) or text.strip()=="":
        return []
    text=text.lower()
    text=re.sub(r'http\S+',' ',text)
    text=re.sub(r'[^a-z\s]',' ',text)
    text=re.sub(r'\s+',' ',text)
    tokens=word_tokenize(text)
    n_grams=list(ngrams(tokens,n))
    return n_grams

def build_bigram_model(corpus):
    bigram_counts=defaultdict(Counter)
    unigram_counts=Counter()
    for text in corpus:
        if not isinstance(text,str):
            continue
        text=text.lower()
        text=re.sub(r'[^a-z\s]',' ',text)
        tokens=word_tokenize(text)
        unigram_counts.update(tokens)
        for w1,w2 in ngrams(tokens,2):
            bigram_counts[w1][w2]+=1
    return bigram_counts,unigram_counts
corpus=df["Resume"].tolist()
bigram_counts,unigram_counts=build_bigram_model(corpus)
print("Vocabulary Size :",len(unigram_counts))
def bigram_probability(w1,w2,bigram_counts,unigram_counts):
    if unigram_counts[w1]==0:
        return 0
    return bigram_counts[w1][w2]/unigram_counts[w1]
print("P('learning'|'machine') =",
      round(bigram_probability('machine','learning',
                               bigram_counts,
                               unigram_counts),4))
print("P('developer'|'software') =",
      round(bigram_probability('software','developer',
                               bigram_counts,
                               unigram_counts),4))
def predict_next_word(word,bigram_counts,top_n=5):
    word=word.lower()
    if word not in bigram_counts:
        return []
    return bigram_counts[word].most_common(top_n)
for word in ["machine","data","python","software"]:
    print("\nNext words after",word)
    print(predict_next_word(word,bigram_counts))
def extract_ngrams(text,n):
    grams=generate_ngrams(text,n)
    return [' '.join(g) for g in grams]
